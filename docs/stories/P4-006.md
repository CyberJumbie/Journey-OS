```python
# POST /solve
class SolveRequest(BaseModel):
    items: list[ItemCandidate]        # candidate pool from item bank
    total_questions: int
    bloom_distribution: dict[int, float]     # {3: 0.4, 4: 0.4, 5: 0.2}
    usmle_system_distribution: dict[str, float] | None   # optional
    difficulty_mix: dict[str, float]          # {"Easy": 0.3, "Medium": 0.5, "Hard": 0.2}
    min_critic_score: float           # 0.0–5.0
    exclude_item_ids: list[str]       # previously used items

class ItemCandidate(BaseModel):
    id: str
    bloom_level: int
    usmle_system: str
    difficulty: str                   # "Easy" | "Medium" | "Hard"
    critic_composite_score: float

class SolveResponse(BaseModel):
    selected_item_ids: list[str]
    solve_time_ms: int
    status: str                       # "optimal" | "infeasible" | "timeout"
    gap_pct: float                    # MIP optimality gap (0 = proven optimal)
    blueprint_validation: dict
```

### MIP Formulation
```python
# Decision variables: x[i] ∈ {0,1} for each item i
prob = LpProblem("exam_assembly", LpMinimize)
x = {item.id: LpVariable(f"x_{item.id}", cat='Binary') for item in items}

# Objective: maximize average critic score of selected items
prob += -lpSum(item.critic_score * x[item.id] for item in items)

# Constraints:
# 1. Total questions
prob += lpSum(x[i] for i in x) == total_questions

# 2. Bloom distribution (soft: allow ±5% tolerance)
for bloom_level, target_pct in bloom_distribution.items():
    bloom_items = [i for i in items if i.bloom_level == bloom_level]
    prob += lpSum(x[i.id] for i in bloom_items) >= floor((target_pct - 0.05) * total_questions)
    prob += lpSum(x[i.id] for i in bloom_items) <= ceil((target_pct + 0.05) * total_questions)

# 3. Difficulty mix (soft: ±5%)
# ... similar pattern ...

# 4. Min critic score
for item in items:
    if item.critic_composite_score < min_critic_score:
        prob += x[item.id] == 0  # hard exclude

# 5. Exclude list
for item_id in exclude_item_ids:
    if item_id in x: prob += x[item_id] == 0

# Solve (timeout 30s)
prob.solve(PULP_CBC_CMD(timeLimit=30, msg=0))
```
